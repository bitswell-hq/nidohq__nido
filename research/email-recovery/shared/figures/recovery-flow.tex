    msg/.style={-{Stealth[length=2mm]}, semithick},

  \draw[msg] (ctrl.north east) -- node[lbl, right=2pt, pos=0.55] {events, votes} (vault.south west);
  \draw[msg] (vault) -- node[lbl, above=1pt] {alert,\\code} (owner);
